if(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT IS_VS6_BUILD)
    # Compile each title's actual observation statements with the real ledger.
    # Startup can revisit collision traversal ordinal zero within one frame.
    foreach(_title IN ITEMS Generals GeneralsMD)
        set(_source "${CMAKE_SOURCE_DIR}/${_title}/Code/GameEngine/Source/GameLogic/Object/PartitionManager.cpp")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_source}")
        file(READ "${_source}" _text)
        string(REPLACE "\r\n" "\n" _text "${_text}")
        string(FIND "${_text}" "void PartitionData::addPossibleCollisions(" _function_start)
        if(_function_start LESS 0)
            message(FATAL_ERROR "${_title} collision traversal entry is missing")
        endif()
        string(SUBSTRING "${_text}" ${_function_start} -1 _function)
        string(FIND "${_function}" "rts::performance::KernelPerformanceBatch performanceBatch =" _begin_start)
        if(_begin_start LESS 0)
            message(FATAL_ERROR "${_title} collision observation entry is missing")
        endif()
        string(SUBSTRING "${_function}" ${_begin_start} -1 _begin)
        string(FIND "${_begin}" ");" _begin_end)
        if(_begin_end LESS 0)
            message(FATAL_ERROR "${_title} collision observation entry is unterminated")
        endif()
        math(EXPR _begin_length "${_begin_end} + 2")
        string(SUBSTRING "${_begin}" 0 ${_begin_length} _begin)
        string(FIND "${_function}" "ownerSnapshot.dirtyOrder" _dirty_start)
        if(_dirty_start LESS 0)
            message(FATAL_ERROR "${_title} collision gameplay order assignment is missing")
        endif()
        string(SUBSTRING "${_function}" ${_dirty_start} -1 _dirty_order)
        string(FIND "${_dirty_order}" ";" _dirty_end)
        if(_dirty_end LESS 0)
            message(FATAL_ERROR "${_title} collision gameplay order assignment is unterminated")
        endif()
        math(EXPR _dirty_length "${_dirty_end} + 1")
        string(SUBSTRING "${_dirty_order}" 0 ${_dirty_length} _dirty_order)
        set(_helper "")
        string(FIND "${_text}" "static rts::JobMetricCounter s_collisionPerformanceOrdinal" _helper_start)
        if(NOT _helper_start LESS 0)
            string(SUBSTRING "${_text}" ${_helper_start} -1 _helper)
            string(FIND "${_helper}" "static void finishCollisionPerformanceBatch(" _helper_end)
            if(_helper_end LESS 0)
                message(FATAL_ERROR "${_title} collision observation helper boundary is missing")
            endif()
            string(SUBSTRING "${_helper}" 0 ${_helper_end} _helper)
        endif()
        set(_generated "${CMAKE_CURRENT_BINARY_DIR}/${_title}")
        file(GENERATE OUTPUT "${_generated}/CollisionPerformanceIdentityTitleHelper.inc" CONTENT "${_helper}\n")
        file(GENERATE OUTPUT "${_generated}/CollisionPerformanceIdentityTitleBegin.inc" CONTENT "${_begin}\n")
        file(GENERATE OUTPUT "${_generated}/CollisionPerformanceIdentityTitleDirtyOrder.inc" CONTENT "${_dirty_order}\n")
        string(TOLOWER "${_title}" _title_lower)
        set(_target "core_collision_performance_identity_${_title_lower}_tests")
        add_executable(${_target} CollisionPerformanceIdentityTitleTest.cpp)
        target_include_directories(${_target} PRIVATE "${_generated}")
        target_compile_definitions(${_target} PRIVATE COLLISION_PERFORMANCE_TITLE="${_title}")
        target_link_libraries(${_target} PRIVATE core_task_runtime)
        set_target_properties(${_target} PROPERTIES
            CXX_STANDARD 20 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
        add_test(NAME ${_target} COMMAND ${_target})
        set_tests_properties(${_target} PROPERTIES TIMEOUT 30)
    endforeach()
endif()
